## 四、HTTP 客户端（api/http.ts）

基于原生 `fetch` 的轻量封装：

```typescript
export const BASE_URL = "http://localhost:8080/api";

// get/post/put/patch/del 都有
async function request<T>(url, options): Promise<T> {
    const response = await fetch(fullUrl, { ...options });

    const data: ApiResponse<T> = await response.json();

    if (data.code !== 200) {
        message.error(data.message);  // Ant Design 全局提示
        throw new Error(data.message);
    }

    return data.data;  // 直接返回 data 字段
}
```

## 五、聊天主界面（AgentChatView.tsx）⭐

### 核心逻辑

```typescript
const AgentChatView: React.FC = () => {
    const { chatSessionId } = useParams();
    const [messages, setMessages] = useState<ChatMessageVO[]>([]);
    const [agentId, setAgentId] = useState<string>("");

    // 1. 加载历史消息
    useEffect(() => {
        if (!chatSessionId) return;
        getChatMessages();  // GET /api/chat-messages/session/{id}
    }, [chatSessionId]);

    // 2. 处理发送消息
    const handleSendMessage = async (message: string) => {
        if (!chatSessionId) {
            // 新会话：先创建会话，再发消息
            const session = await createChatSession({ agentId, title });
            await createChatMessage({ agentId, sessionId: session.chatSessionId, ... });
            navigate(`/chat/${session.chatSessionId}`);
        } else {
            // 已有会话：直接发消息
            await createChatMessage({ sessionId: chatSessionId, ... });
            await getChatMessages();  // 刷新消息列表
        }
    };

    // 3. ⭐ SSE 连接
    useEffect(() => {
        if (!chatSessionId) return;

        const es = new EventSource(`http://localhost:8080/sse/connect/${chatSessionId}`);

        es.addEventListener("message", (event) => {
            const msg = JSON.parse(event.data) as SseMessage;
            if (msg.type === "AI_GENERATED_CONTENT") {
                addMessage(msg.payload.message);          // 追加新消息
            } else {
                // 显示 Agent 状态（思考中/执行中/完成）
                setDisplayAgentStatus(...);
            }
        });

        return () => es.close();  // 清理
    }, [chatSessionId]);

    return <div>
        <AgentChatHistory messages={messages} ... />
        <AgentChatInput onSend={handleSendMessage} />
    </div>;
};
```

## 八、面试核心问题

### Q1: 为什么选 fetch 而不是 axios？

轻量，不需要额外依赖。封装了统一的错误处理和类型推导。

### Q2: SSE 前端连接怎么处理的？

`useEffect` 中创建 `EventSource`，`addEventListener("message")` 监听，组件卸载时 `es.close()`。

### Q3: 自动滚动怎么实现的？

滚动容器 ref，监听 scroll 事件判断是否在底部（20px 阈值），有新消息且用户在底部时自动滚到底部。
